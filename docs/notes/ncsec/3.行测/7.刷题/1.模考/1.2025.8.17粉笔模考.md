### **3. 题型数据看板**

::: echarts

```js
const option = {
  tooltip: {
    trigger: "axis",
    axisPointer: { type: "shadow" },
    // 自定义悬浮内容（核心：显示正确率）
    formatter: function (params) {
      // params 是数组，每个元素对应一条数据（这里只有 1 个系列，取第 0 个）
      const item = params[0];
      // 从 y 轴数据中拆分出 “题型名称” 和 “正确率”
      const [name, rate] = item.name.split(" ");
      return `
        题型：${name}<br>
        用时：${item.value} 秒<br>
        正确率：${rate}
      `;
    },
  },
  grid: {
    left: "12%",
    right: "8%",
    bottom: "15%",
    containLabel: true,
  },
  xAxis: {
    type: "value",
    name: "用时（秒）",
    nameTextStyle: { fontSize: 12, padding: [5, 0, 0, 0] },
    axisLabel: { fontSize: 12 },
  },
  yAxis: {
    type: "category",
    // Y 轴数据格式："题型名称 正确率"
    data: [
      "政治理论 35%",
      "常识判断 47%",
      "言语理解与表达 77%",
      "数量关系 90%",
      "判断推理 71%",
      "资料分析 15%",
    ],
    axisLabel: {
      fontSize: 12,
      // 强制换行（让文字更规整）
      formatter: (value) => value.split(" ").join("\n"),
      margin: 15,
    },
    axisLine: { show: false },
    axisTick: { show: false },
  },
  series: [
    {
      name: "用时",
      type: "bar",
      data: [480, 480, 1560, 2460, 2040, 34],
      barWidth: "40%",
      itemStyle: { color: "#127443ff" },
      label: {
        show: true,
        position: "right",
        color: "#333",
        fontSize: 12,
      },
    },
  ],
};
```
